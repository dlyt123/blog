-- 1. post 表：新增 author_id（文章作者，用于区分「管理员 / 普通用户」的文章归属）
SET @has := (SELECT COUNT(*) FROM information_schema.COLUMNS
             WHERE TABLE_SCHEMA = DATABASE() AND TABLE_NAME = 'post' AND COLUMN_NAME = 'author_id');
SET @ddl := IF(@has = 0,
  'ALTER TABLE `post` ADD COLUMN `author_id` BIGINT DEFAULT NULL COMMENT ''作者ID（关联user表）''',
  'SELECT 1');
PREPARE stmt FROM @ddl; EXECUTE stmt; DEALLOCATE PREPARE stmt;

-- 2. comment 表：新增 user_id（评论者用户ID；昵称头像改为从这里实时联表读取）
SET @has := (SELECT COUNT(*) FROM information_schema.COLUMNS
             WHERE TABLE_SCHEMA = DATABASE() AND TABLE_NAME = 'comment' AND COLUMN_NAME = 'user_id');
SET @ddl := IF(@has = 0,
  'ALTER TABLE `comment` ADD COLUMN `user_id` BIGINT DEFAULT NULL COMMENT ''评论者用户ID（关联user表）''',
  'SELECT 1');
PREPARE stmt FROM @ddl; EXECUTE stmt; DEALLOCATE PREPARE stmt;

-- 3. comment.status 默认值改为 1（评论免审核，发表后直接可见；管理员仍可驳回/删除）
ALTER TABLE `comment`
  MODIFY COLUMN `status` TINYINT NULL DEFAULT 1
  COMMENT '0待审核 1通过 2驳回（默认直接通过，免审核）';

-- 4. post_like：点赞记录表（一个用户对一篇文章只能点一次）
CREATE TABLE IF NOT EXISTS `post_like` (
  `id`          BIGINT   NOT NULL AUTO_INCREMENT,
  `user_id`     BIGINT   NOT NULL,
  `post_id`     BIGINT   NOT NULL,
  `create_time` DATETIME DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  UNIQUE KEY `uk_user_post` (`user_id`, `post_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 5. post_favorite：收藏记录表（一个用户对一篇文章只能收藏一次）
CREATE TABLE IF NOT EXISTS `post_favorite` (
  `id`          BIGINT   NOT NULL AUTO_INCREMENT,
  `user_id`     BIGINT   NOT NULL,
  `post_id`     BIGINT   NOT NULL,
  `create_time` DATETIME DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  UNIQUE KEY `uk_user_post` (`user_id`, `post_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 6. user.role：普通用户默认角色改为 USER（只有管理员才是 ADMIN）
ALTER TABLE `user`
  MODIFY COLUMN `role` VARCHAR(20) DEFAULT 'USER'
  COMMENT 'USER普通用户(博主) / ADMIN管理员';

-- 7. visit_log：访客留痕表（记录每一次页面访问，含未登录游客）
CREATE TABLE IF NOT EXISTS `visit_log` (
  `id`          BIGINT       NOT NULL AUTO_INCREMENT,
  `user_id`     BIGINT       DEFAULT NULL COMMENT '访问者用户ID；为空表示未登录游客',
  `ip`          VARCHAR(64)  DEFAULT NULL,
  `path`        VARCHAR(255) DEFAULT NULL COMMENT '访问的前端路由，如 /posts/1',
  `referer`     VARCHAR(500) DEFAULT NULL COMMENT '来源页面',
  `user_agent`  VARCHAR(500) DEFAULT NULL,
  `device`      VARCHAR(100) DEFAULT NULL COMMENT '浏览器 / 系统',
  `create_time` DATETIME     DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  KEY `idx_create_time` (`create_time`),
  KEY `idx_user_id` (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='访客访问留痕，仅管理员可查看';

-- 8. sensitive_word：评论敏感词库（管理员可在后台「敏感词管理」页维护）
CREATE TABLE IF NOT EXISTS `sensitive_word` (
  `id`          BIGINT      NOT NULL AUTO_INCREMENT,
  `word`        VARCHAR(64) NOT NULL COMMENT '敏感词',
  `create_time` DATETIME    DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  UNIQUE KEY `uk_word` (`word`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='评论敏感词库（管理员可维护）';

-- 9. user_follow：用户关注关系（用于「用户主页 + 关注」功能）
CREATE TABLE IF NOT EXISTS `user_follow` (
  `id`          BIGINT   NOT NULL AUTO_INCREMENT,
  `follower_id` BIGINT   NOT NULL COMMENT '关注者（谁点的关注）',
  `followee_id` BIGINT   NOT NULL COMMENT '被关注者（被关注的人）',
  `create_time` DATETIME DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  UNIQUE KEY `uk_follow` (`follower_id`, `followee_id`),

-- 10. sensitive_log：敏感词命中记录（管理员后台可查看被拦了什么）
CREATE TABLE IF NOT EXISTS `sensitive_log` (
  `id`          BIGINT       NOT NULL AUTO_INCREMENT,
  `word`        VARCHAR(64)  NOT NULL COMMENT '命中的敏感词',
  `scene`       VARCHAR(50)  COMMENT '触发场景',
  `content`     VARCHAR(1000) COMMENT '被拦截的内容（截断存储）',
  `create_time` DATETIME     DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  KEY `idx_time` (`create_time`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='敏感词命中记录（仅管理员可见）';

-- 11. report：内容举报
CREATE TABLE IF NOT EXISTS `report` (
  `id`          BIGINT       NOT NULL AUTO_INCREMENT,
  `target_type` VARCHAR(20)  NOT NULL COMMENT 'post 文章 / comment 评论',
  `target_id`   BIGINT       NOT NULL,
  `reason`      VARCHAR(200) COMMENT '举报理由',
  `user_id`     BIGINT       COMMENT '举报人 id',
  `status`      TINYINT      DEFAULT 0 COMMENT '0 待处理 / 1 已处理',
  `create_time` DATETIME     DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  KEY `idx_status` (`status`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='内容举报（管理员后台处理）';

-- 12. series：文章系列 / 专栏
CREATE TABLE IF NOT EXISTS `series` (
  `id`          BIGINT       NOT NULL AUTO_INCREMENT,
  `name`        VARCHAR(100) NOT NULL,
  `description` VARCHAR(500),
  `sort`        INT          DEFAULT 0,
  `create_time` DATETIME     DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='文章系列 / 专栏';

-- 13. post 增加 series_id 列（幂等）
SET @col_exists := (SELECT COUNT(*) FROM information_schema.columns

SET @ddl := IF(@col_exists = 0,
  'ALTER TABLE post ADD COLUMN series_id BIGINT DEFAULT NULL COMMENT ''所属系列'' AFTER category_id',

-- 15. operation_log：后台操作日志（审计）
CREATE TABLE IF NOT EXISTS `operation_log` (
  `id`          BIGINT       NOT NULL AUTO_INCREMENT,
  `user_id`     BIGINT       COMMENT '操作人 id',
  `username`    VARCHAR(64)  COMMENT '操作人用户名',
  `action`      VARCHAR(50)  COMMENT '动作：新增/修改/删除',
  `target`      VARCHAR(200) COMMENT '操作对象（如：文章 #3）',
  `method`      VARCHAR(10)  COMMENT 'HTTP 方法',
  `path`        VARCHAR(200) COMMENT '请求路径',
  `ip`          VARCHAR(64),
  `success`     TINYINT      DEFAULT 1 COMMENT '1 成功 / 0 失败',
  `create_time` DATETIME     DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  KEY `idx_time` (`create_time`),
  KEY `idx_user` (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='后台操作日志（审计）';

-- 20. message：私信
CREATE TABLE IF NOT EXISTS `message` (
  `id`           BIGINT        NOT NULL AUTO_INCREMENT,
  `from_user_id` BIGINT        NOT NULL COMMENT '发送者',
  `to_user_id`   BIGINT        NOT NULL COMMENT '接收者',
  `content`      VARCHAR(1000) NOT NULL COMMENT '内容',
  `is_read`      TINYINT       DEFAULT 0 COMMENT '1=已读',
  `create_time`  DATETIME      DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  KEY `idx_from` (`from_user_id`),
  KEY `idx_to_read` (`to_user_id`, `is_read`),
  KEY `idx_pair` (`from_user_id`, `to_user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='私信';

-- 【第九批：性能优化 —— 索引补齐】
--
-- 背景：用 EXPLAIN 实测发现 9 处查询没有合适索引，会出现
--       Using filesort（排序要落盘）或 type=ALL（整表扫描）。
--       当前数据量小（文章 12 篇）感觉不出来，但这是**为上线后数据增长做准备**。
--
-- ⚠️ 全部用 information_schema 判断后再执行，可重复运行。
-- ================================================================

-- 49. post：首页列表（最高频）
--     ⚠️ 列顺序是照着**真实 SQL** 定的：首页排序是 `pinned DESC, publish_time DESC`，
--        所以 pinned 必须排在 publish_time 前面，否则索引覆盖不了排序、照样 Using filesort。
--     实测：EXPLAIN 显示 key=idx_list、Extra=Using index（无 filesort）。
SET @c := (SELECT COUNT(*) FROM information_schema.STATISTICS
  WHERE table_schema='blog' AND table_name='post' AND index_name='idx_list');
SET @ddl := IF(@c = 0,
  'ALTER TABLE post ADD INDEX idx_list (deleted, status, pinned, publish_time)',
  'SELECT ''post.idx_list 已存在''');
PREPARE s FROM @ddl; EXECUTE s; DEALLOCATE PREPARE s;

-- 50. post：用户主页 / 作者页
--     真实 SQL 是 `WHERE deleted=0 AND status=1 AND author_id=? ORDER BY id DESC`，
--     排序键 id 是主键、被二级索引隐含带上，所以这三列就够（实测无 filesort）。
SET @c := (SELECT COUNT(*) FROM information_schema.STATISTICS
  WHERE table_schema='blog' AND table_name='post' AND index_name='idx_author');
SET @ddl := IF(@c = 0,
  'ALTER TABLE post ADD INDEX idx_author (author_id, deleted, status)',
  'SELECT ''post.idx_author 已存在''');
PREPARE s FROM @ddl; EXECUTE s; DEALLOCATE PREPARE s;

-- 51. post：系列页（排序是 publish_time ASC，所以 publish_time 要在最后）
